## 1. 概述

### 1.1 设计目标

- **自动化**：定时采集，无需人工干预
- **多源聚合**：GitHub、HuggingFace、OpenRouter、Reddit、Twitter
- **增量更新**：只爬取新数据，避免重复
- **高质量**：数据清洗、去重、标准化
- **可扩展**：易于添加新数据源

### 1.2 设计原则

1. **只写原始数据** — 爬虫不计算、不做业务逻辑，只负责采集和去重后写入 raw_data 表
2. **严格遵守 robots.txt** — 优先使用官方 API
3. **幂等写入** — 同一数据重复抓取时覆盖而非堆积
4. **异常隔离** — 单个爬虫出错不影响其他爬虫
5. **可追溯** — 所有操作写入日志，便于排查

### 1.3 数据流

```
数据源 → 爬虫 → 清洗 → 去重 → 入库(raw_data) → AI生成
```

### 3.2 基类设计

```python
# core/base_spider.py

from abc import ABC, abstractmethod
import hashlib
import json
from datetime import datetime
from typing import Dict, List, Optional

class BaseSpider(ABC):
    """爬虫基类"""
    
    def __init__(self, source: str, data_type: str):
        self.source = source
        self.data_type = data_type
        self.db = get_db_connection()
    
    @abstractmethod
    def fetch_data(self) -> List[Dict]:
        """抓取数据（子类实现）"""
        pass
    
    def run(self):
        """运行爬虫"""
        try:
            # 1. 抓取数据
            raw_items = self.fetch_data()
            print(f"[{self.source}] 抓取到 {len(raw_items)} 条数据")
            
            # 2. 处理数据
            processed_items = []
            for item in raw_items:
                processed = self.process_item(item)
                if processed:
                    processed_items.append(processed)
            
            # 3. 去重
            unique_items = self.deduplicate(processed_items)
            print(f"[{self.source}] 去重后 {len(unique_items)} 条数据")
            
            # 4. 写入数据库
            self.save_to_db(unique_items)
            print(f"[{self.source}] 写入成功")
            
            return len(unique_items)
        
        except Exception as e:
            print(f"[{self.source}] 错误: {e}")
            self.log_error(e)
            return 0
    
    def process_item(self, item: Dict) -> Optional[Dict]:
        """处理单条数据"""
        try:
            # 生成唯一ID
            source_id = self.generate_source_id(item)
            
            # 生成内容哈希
            content_hash = self.generate_content_hash(item)
            
            return {
                "source": self.source,
                "data_type": self.data_type,
                "source_id": source_id,
                "title": item.get('title', ''),
                "description": item.get('description', ''),
                "url": item.get('url', ''),
                "metadata": item.get('metadata', {}),
                "raw_content": json.dumps(item, ensure_ascii=False),
                "content_hash": content_hash,
                "fetched_at": datetime.utcnow().isoformat()
            }
        except Exception as e:
            print(f"处理数据失败: {e}")
            return None
    
    def generate_source_id(self, item: Dict) -> str:
        """生成唯一ID（子类可覆盖）"""
        return item.get('id', item.get('url', ''))
    
    def generate_content_hash(self, item: Dict) -> str:
        """生成内容哈希"""
        content = json.dumps(item, sort_keys=True, ensure_ascii=False)
        return hashlib.md5(content.encode()).hexdigest()
    
    def deduplicate(self, items: List[Dict]) -> List[Dict]:
        """去重"""
        # 1. 内存去重（同一批次）
        seen = set()
        unique = []
        for item in items:
            key = f"{item['source']}:{item['source_id']}"
            if key not in seen:
                seen.add(key)
                unique.append(item)
        
        # 2. 数据库去重（检查是否已存在）
        final = []
        for item in unique:
            existing = self.db.table('raw_data').select('content_hash').eq('source', item['source']).eq('source_id', item['source_id']).execute()
            
            if not existing.data:
                # 不存在，添加
                final.append(item)
            elif existing.data[0]['content_hash'] != item['content_hash']:
                # 存在但内容变化，更新
                final.append(item)
        
        return final
    
    def save_to_db(self, items: List[Dict]):
        """写入数据库"""
        if not items:
            return
        
        # 使用 upsert（存在则更新，不存在则插入）
        self.db.table('raw_data').upsert(items, on_conflict='source,source_id').execute()
    
    def log_error(self, error: Exception):
        """记录错误"""
        self.db.table('spider_logs').insert({
            "source": self.source,
            "data_type": self.data_type,
            "status": "error",
            "message": str(error),
            "created_at": datetime.utcnow().isoformat()
        }).execute()
```

### 3.3 GitHub爬虫实现

```python
# spiders/github_spider.py

import requests
from core.base_spider import BaseSpider
from config import GITHUB_TOKEN

class GitHubSpider(BaseSpider):
    def __init__(self):
        super().__init__(source='github', data_type='repo')
        self.api_url = "https://api.github.com"
        self.headers = {
            "Authorization": f"token {GITHUB_TOKEN}",
            "Accept": "application/vnd.github.v3+json"
        }
    
    def fetch_data(self):
        """抓取 GitHub 仓库"""
        repos = []
        
        # 搜索 AI 相关仓库
        queries = [
            "topic:artificial-intelligence",
            "topic:machine-learning",
            "topic:deep-learning",
            "topic:llm"
        ]
        
        for query in queries:
            url = f"{self.api_url}/search/repositories"
            params = {
                "q": query,
                "sort": "stars",
                "order": "desc",
                "per_page": 50
            }
            
            response = requests.get(url, headers=self.headers, params=params)
            response.raise_for_status()
            
            data = response.json()
            repos.extend(data['items'])
        
        return repos
    
    def process_item(self, repo):
        """处理 GitHub 仓库数据"""
        return {
            "id": repo['full_name'],
            "title": repo['name'],
            "description": repo.get('description', ''),
            "url": repo['html_url'],
            "metadata": {
                "stars": repo['stargazers_count'],
                "forks": repo['forks_count'],
                "language": repo.get('language'),
                "topics": repo.get('topics', []),
                "created_at": repo['created_at'],
                "updated_at": repo['updated_at']
            }
        }
```

## 4. 数据质量保证

### 4.1 数据验证

```python
def validate_raw_data(item: Dict) -> bool:
    """验证数据完整性"""
    required_fields = ['source', 'data_type', 'source_id', 'title', 'url']
    
    for field in required_fields:
        if not item.get(field):
            print(f"缺少必填字段: {field}")
            return False
    
    # URL格式验证
    if not item['url'].startswith('http'):
        print(f"URL格式错误: {item['url']}")
        return False
    
    return True
```

def safe_request(url, max_retries=3):
    """安全的HTTP请求（带重试）"""
    for i in range(max_retries):
        try:
            response = requests.get(url, timeout=10)
            response.raise_for_status()
            return response
        except requests.exceptions.RequestException as e:
