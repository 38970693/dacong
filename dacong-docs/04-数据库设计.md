# 数据库设计

> 版本：v2.0  
> 更新日期：2026-04-17  
> 数据库：PostgreSQL 15（Supabase）

---

## 1. 设计原则

### 1.1 核心原则

1. **多语言支持**：单表多字段方案（title_zh、title_en）
2. **数据分层**：raw_data（原始）→ generated_content（待审核）→ published_content（已发布）
3. **审计追踪**：记录创建时间、更新时间、操作人
4. **性能优先**：合理使用索引、JSONB字段
5. **安全第一**：RLS策略保护数据

### 1.2 命名规范

- 表名：小写 + 下划线（snake_case）
- 字段名：小写 + 下划线
- 主键：统一使用UUID
- 时间戳：统一使用TIMESTAMPTZ
- 布尔值：is_xxx、has_xxx、can_xxx

## 3. 表结构设计

### 3.1 raw_data（爬虫原始数据）

**用途：** 存储爬虫采集的原始数据，未经处理

```sql
CREATE TABLE raw_data (
  -- 主键
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 数据来源
  source TEXT NOT NULL,  -- 'github', 'huggingface', 'openrouter', 'reddit', 'twitter'
  data_type TEXT NOT NULL,  -- 'model', 'tool', 'price', 'discussion', 'news'
  
  -- 原始数据（JSONB格式，灵活存储）
  raw_json JSONB NOT NULL,
  
  -- 元数据
  fetched_at TIMESTAMPTZ DEFAULT NOW(),  -- 爬取时间
  processed BOOLEAN DEFAULT FALSE,  -- 是否已处理
  processed_at TIMESTAMPTZ,  -- 处理时间
  
  -- 审计字段
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_raw_data_source ON raw_data(source, fetched_at DESC);
CREATE INDEX idx_raw_data_processed ON raw_data(processed, created_at DESC) WHERE NOT processed;
CREATE INDEX idx_raw_data_type ON raw_data(data_type, source);

-- 注释
COMMENT ON TABLE raw_data IS '爬虫采集的原始数据';
COMMENT ON COLUMN raw_data.source IS '数据来源：github/huggingface/openrouter等';
COMMENT ON COLUMN raw_data.data_type IS '数据类型：model/tool/price等';
COMMENT ON COLUMN raw_data.raw_json IS '原始JSON数据，结构由source决定';
```

### 3.2 generated_content（AI生成内容）

**用途：** 存储AI生成的内容，等待人工审核

```sql
CREATE TABLE generated_content (
  -- 主键
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 内容类型
  content_type TEXT NOT NULL,  -- 'recommendation', 'guide', 'comparison', 'weekly'
  
  -- 中文内容（必填，主战场）
  title_zh TEXT NOT NULL,
  summary_zh TEXT,
  content_zh JSONB NOT NULL,
  
  -- 英文内容（AI翻译，副战场）
  title_en TEXT,
  summary_en TEXT,
  content_en JSONB,
  
  -- 关联数据
  source_data_ids UUID[],  -- 关联的raw_data ID数组
  scenario TEXT,  -- 场景：'coding', 'video', 'image', 'chat'
  
  -- AI质量指标
  ai_confidence FLOAT CHECK (ai_confidence >= 0 AND ai_confidence <= 1),  -- AI自评置信度
  quality_score FLOAT CHECK (quality_score >= 0 AND quality_score <= 1),  -- 自动质量评分
  
  -- 审核状态
  review_status TEXT DEFAULT 'pending' CHECK (review_status IN ('pending', 'approved', 'rejected', 'editing')),
  review_notes TEXT,  -- 审核备注
  
  -- 时间戳
  generated_at TIMESTAMPTZ DEFAULT NOW(),
  reviewed_at TIMESTAMPTZ,
  reviewed_by UUID,  -- 审核人（关联auth.users）
  
  -- 审计字段
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_generated_content_status ON generated_content(review_status, generated_at DESC);
CREATE INDEX idx_generated_content_type ON generated_content(content_type, review_status);
CREATE INDEX idx_generated_content_scenario ON generated_content(scenario) WHERE scenario IS NOT NULL;
CREATE INDEX idx_generated_content_quality ON generated_content(quality_score DESC, ai_confidence DESC);

-- 注释
COMMENT ON TABLE generated_content IS 'AI生成的内容，等待人工审核';
COMMENT ON COLUMN generated_content.content_zh IS '中文内容（JSONB），结构由content_type决定';
COMMENT ON COLUMN generated_content.content_en IS '英文内容（AI翻译）';
COMMENT ON COLUMN generated_content.ai_confidence IS 'AI自评置信度（0-1），低于0.85需人工审核';
COMMENT ON COLUMN generated_content.quality_score IS '自动质量评分（0-1），综合多个维度';
```

### 3.3 published_content（已发布内容）

**用途：** 存储已发布的内容，供客户端展示

```sql
CREATE TABLE published_content (
  -- 主键
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 关联
  generated_content_id UUID REFERENCES generated_content(id) ON DELETE SET NULL,
  
  -- 内容类型
  content_type TEXT NOT NULL,
  
  -- 中文内容
  title_zh TEXT NOT NULL,
  summary_zh TEXT,
  content_zh JSONB NOT NULL,
  
  -- 英文内容
  title_en TEXT,
  summary_en TEXT,
  content_en JSONB,
  
  -- 场景
  scenario TEXT,
  
  -- 统计数据
  view_count INT DEFAULT 0 CHECK (view_count >= 0),
  helpful_count INT DEFAULT 0 CHECK (helpful_count >= 0),
  unhelpful_count INT DEFAULT 0 CHECK (unhelpful_count >= 0),
  share_count INT DEFAULT 0 CHECK (share_count >= 0),
  
  -- 时间戳
  published_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- 审计字段
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_published_content_type ON published_content(content_type, published_at DESC);
CREATE INDEX idx_published_content_scenario ON published_content(scenario, published_at DESC) WHERE scenario IS NOT NULL;
CREATE INDEX idx_published_content_popular ON published_content(helpful_count DESC, view_count DESC);
CREATE INDEX idx_published_content_updated ON published_content(updated_at DESC);

-- 全文搜索索引（中文）
CREATE INDEX idx_published_content_search_zh ON published_content USING gin(to_tsvector('simple', title_zh || ' ' || COALESCE(summary_zh, '')));

-- 全文搜索索引（英文）
CREATE INDEX idx_published_content_search_en ON published_content USING gin(to_tsvector('english', COALESCE(title_en, '') || ' ' || COALESCE(summary_en, '')));

-- 注释
COMMENT ON TABLE published_content IS '已发布的内容，供客户端展示';
COMMENT ON COLUMN published_content.helpful_count IS '用户点击"有用"的次数';
COMMENT ON COLUMN published_content.unhelpful_count IS '用户点击"无用"的次数';
```

### 3.4 user_feedback（用户反馈）

**用途：** 收集用户对内容的反馈

```sql
CREATE TABLE user_feedback (
  -- 主键
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 关联
  content_id UUID NOT NULL REFERENCES published_content(id) ON DELETE CASCADE,
  user_id UUID,  -- 可选，匿名用户为NULL，关联auth.users
  
  -- 反馈内容
  is_helpful BOOLEAN NOT NULL,  -- true=有用，false=无用
  comment TEXT,  -- 可选的文字评论
  
  -- 元数据
  user_agent TEXT,  -- 浏览器信息
  ip_address INET,  -- IP地址（用于去重）
  
  -- 时间戳
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 索引
CREATE INDEX idx_user_feedback_content ON user_feedback(content_id, created_at DESC);
CREATE INDEX idx_user_feedback_user ON user_feedback(user_id, created_at DESC) WHERE user_id IS NOT NULL;
CREATE INDEX idx_user_feedback_helpful ON user_feedback(is_helpful, created_at DESC);

-- 唯一约束（防止重复反馈）
CREATE UNIQUE INDEX idx_user_feedback_unique_user ON user_feedback(content_id, user_id) WHERE user_id IS NOT NULL;
CREATE UNIQUE INDEX idx_user_feedback_unique_anon ON user_feedback(content_id, ip_address) WHERE user_id IS NULL;

-- 注释
COMMENT ON TABLE user_feedback IS '用户对内容的反馈';
COMMENT ON COLUMN user_feedback.is_helpful IS 'true=有用，false=无用';
```

### 3.5 scenario_recommendations（场景推荐配置）

**用途：** 配置每个场景的推荐内容顺序

```sql
CREATE TABLE scenario_recommendations (
  -- 主键
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- 场景
  scenario TEXT NOT NULL,  -- 'coding', 'video', 'image', 'chat', 'writing', 'data'
  
  -- 排序
  rank INT NOT NULL CHECK (rank > 0),
  
  -- 关联内容
  content_id UUID NOT NULL REFERENCES published_content(id) ON DELETE CASCADE,
  
  -- 状态
  active BOOLEAN DEFAULT TRUE,
  
  -- 时间戳
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- 唯一约束
  UNIQUE(scenario, rank)
);

-- 索引
CREATE INDEX idx_scenario_recommendations ON scenario_recommendations(scenario, rank) WHERE active;
CREATE INDEX idx_scenario_recommendations_content ON scenario_recommendations(content_id);

-- 注释
COMMENT ON TABLE scenario_recommendations IS '场景推荐配置表';
COMMENT ON COLUMN scenario_recommendations.rank IS '排序（1=最佳，2=平替，3=免费）';
```

## 4. Row Level Security (RLS)

### 4.1 安全策略

```sql
-- 启用RLS
ALTER TABLE raw_data ENABLE ROW LEVEL SECURITY;
ALTER TABLE generated_content ENABLE ROW LEVEL SECURITY;
ALTER TABLE published_content ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_feedback ENABLE ROW LEVEL SECURITY;
ALTER TABLE scenario_recommendations ENABLE ROW LEVEL SECURITY;

-- raw_data：只有服务端可以访问
CREATE POLICY "Service role can do everything on raw_data"
ON raw_data FOR ALL
TO service_role
USING (true)
WITH CHECK (true);

-- generated_content：只有管理员可以访问
CREATE POLICY "Admins can do everything on generated_content"
ON generated_content FOR ALL
TO authenticated
USING (auth.jwt() ->> 'role' = 'admin')
WITH CHECK (auth.jwt() ->> 'role' = 'admin');

-- published_content：所有人可以查看
CREATE POLICY "Anyone can view published_content"
ON published_content FOR SELECT
USING (true);

-- published_content：只有管理员可以修改
CREATE POLICY "Admins can modify published_content"
ON published_content FOR ALL
TO authenticated
USING (auth.jwt() ->> 'role' = 'admin')
WITH CHECK (auth.jwt() ->> 'role' = 'admin');

-- user_feedback：所有人可以创建
CREATE POLICY "Anyone can create feedback"
ON user_feedback FOR INSERT
WITH CHECK (true);

-- user_feedback：用户可以查看自己的反馈
CREATE POLICY "Users can view own feedback"
ON user_feedback FOR SELECT
TO authenticated
USING (auth.uid() = user_id);

-- user_feedback：管理员可以查看所有反馈
CREATE POLICY "Admins can view all feedback"
ON user_feedback FOR SELECT
TO authenticated
USING (auth.jwt() ->> 'role' = 'admin');

-- scenario_recommendations：所有人可以查看
CREATE POLICY "Anyone can view scenario_recommendations"
ON scenario_recommendations FOR SELECT
USING (active = true);

-- scenario_recommendations：只有管理员可以修改
CREATE POLICY "Admins can modify scenario_recommendations"
ON scenario_recommendations FOR ALL
TO authenticated
USING (auth.jwt() ->> 'role' = 'admin')
WITH CHECK (auth.jwt() ->> 'role' = 'admin');
```

## 5. 触发器与函数

### 5.1 自动更新updated_at

```sql
-- 创建触发器函数
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 应用到表
CREATE TRIGGER update_generated_content_updated_at
BEFORE UPDATE ON generated_content
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_published_content_updated_at
BEFORE UPDATE ON published_content
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_scenario_recommendations_updated_at
BEFORE UPDATE ON scenario_recommendations
FOR EACH ROW
EXECUTE FUNCTION update_updated_at_column();
```

### 5.2 自动更新统计数据

```sql
-- 用户反馈时自动更新published_content的统计
CREATE OR REPLACE FUNCTION update_content_stats()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.is_helpful THEN
    UPDATE published_content
    SET helpful_count = helpful_count + 1
    WHERE id = NEW.content_id;
  ELSE
    UPDATE published_content
    SET unhelpful_count = unhelpful_count + 1
    WHERE id = NEW.content_id;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_content_stats_on_feedback
AFTER INSERT ON user_feedback
FOR EACH ROW
EXECUTE FUNCTION update_content_stats();
```

## 6. 视图

### 6.1 内容统计视图

```sql
CREATE OR REPLACE VIEW content_stats AS
SELECT
  pc.id,
  pc.content_type,
  pc.title_zh,
  pc.scenario,
  pc.view_count,
  pc.helpful_count,
  pc.unhelpful_count,
  pc.share_count,
  -- 计算有用率
  CASE
    WHEN (pc.helpful_count + pc.unhelpful_count) > 0
    THEN ROUND(pc.helpful_count::NUMERIC / (pc.helpful_count + pc.unhelpful_count), 2)
    ELSE NULL
  END AS helpful_rate,
  pc.published_at,
  pc.updated_at
FROM published_content pc
ORDER BY pc.published_at DESC;

COMMENT ON VIEW content_stats IS '内容统计视图';
```

### 6.2 待审核队列视图

```sql
CREATE OR REPLACE VIEW review_queue AS
SELECT
  gc.id,
  gc.content_type,
  gc.title_zh,
  gc.scenario,
  gc.ai_confidence,
  gc.quality_score,
  gc.review_status,
  gc.generated_at,
  -- 计算优先级分数（低置信度优先）
  CASE
    WHEN gc.ai_confidence < 0.85 THEN 10
    WHEN gc.quality_score < 0.8 THEN 8
    ELSE 0
  END AS priority_score,
  gc.created_at
FROM generated_content gc
WHERE gc.review_status = 'pending'
ORDER BY priority_score DESC, gc.generated_at ASC;

COMMENT ON VIEW review_queue IS '待审核队列视图（按优先级排序）';
```

## 8. 性能优化

### 8.1 查询优化

**常用查询1：获取场景推荐**
```sql
-- 优化前（慢）
SELECT * FROM published_content
WHERE scenario = 'coding'
ORDER BY published_at DESC;

-- 优化后（快）
SELECT pc.*
FROM scenario_recommendations sr
JOIN published_content pc ON sr.content_id = pc.id
WHERE sr.scenario = 'coding' AND sr.active = true
ORDER BY sr.rank;

-- 使用索引：idx_scenario_recommendations
```

**常用查询2：搜索内容**
```sql
-- 中文搜索
SELECT *
FROM published_content
WHERE to_tsvector('simple', title_zh || ' ' || COALESCE(summary_zh, ''))
  @@ to_tsquery('simple', 'AI & 工具')
ORDER BY published_at DESC
LIMIT 20;

-- 使用索引：idx_published_content_search_zh
```

### 8.2 JSONB优化
