## 1. 概述

### 1.1 设计目标

- **自动化**：爬虫采集 → AI生成 → 自动发布，减少人工成本
- **高质量**：多轮生成 + 质量检测，确保内容质量
- **多语言**：中文（主战场）+ 英文（AI翻译）
- **低成本**：使用Prompt Caching，月成本控制在$50-150

### 1.2 生成流程

```
1. 数据准备
   从raw_data表读取 → 数据清洗 → 结构化

2. Prompt构建
   系统Prompt（缓存）+ Few-shot样本（缓存）+ 用户数据（不缓存）

3. 多轮生成
   初稿 → 自我批评 → 改进版

4. 质量检测
   完整性 + 敏感词 + 连贯性 → 计算质量分数

5. 翻译
   中文 → 英文（AI自动翻译）

6. 入库
   写入generated_content表 → 触发审核
```

```python
def generate_draft(scenario, data):
```

```python
def critique_draft(draft):

    response = claude_api.messages.create(
```

```python
def generate_content(scenario, data):
    # 第一轮：生成初稿
    draft = generate_draft(scenario, data)
    
    # 第二轮：自我批评
    critique = critique_draft(draft)
    
    # 第三轮：改进版本
    final = improve_draft(draft, critique)
    
    # 质量检测
    quality_score = check_quality(final)
    
    return {
        "content": final,
        "quality_score": quality_score,
        "ai_confidence": final.get("confidence", 0.5)
    }
```

```python
def check_quality(content):
    scores = {}
    
    # 1. 完整性检查（30分）
    scores['completeness'] = check_completeness(content)
    
    # 2. 敏感词检查（20分）
    scores['safety'] = check_sensitive_words(content)
    
    # 3. 逻辑连贯性（20分）
    scores['coherence'] = check_coherence(content)
    
    # 4. 数据支撑（15分）
    scores['data_backed'] = check_data_support(content)
    
    # 5. 用户友好（15分）
    scores['user_friendly'] = check_user_friendly(content)
    
    # 总分
    total_score = sum(scores.values()) / 100
    
    return {
        "total_score": total_score,
        "details": scores
    }
```

### 4.2 完整性检查

```python
def check_completeness(content):
    score = 0
    
    # 必填字段
    required_fields = ['title', 'summary', 'recommendations']
    for field in required_fields:
        if field in content and content[field]:
            score += 10
    
    # 推荐数量（至少3个）
    if len(content.get('recommendations', [])) >= 3:
        score += 10
    
    # 每个推荐的完整性
    for rec in content.get('recommendations', []):
        if all(k in rec for k in ['name', 'pros', 'cons', 'price', 'access']):
            score += 5
            break
    
    return min(score, 30)
```

### 4.3 敏感词检查

```python
SENSITIVE_WORDS = [
    '政治', '习近平', '共产党', '台独', '藏独',
    '法轮功', '六四', '天安门', '翻墙', 'VPN',
    # ... 更多敏感词
]

def check_sensitive_words(content):
    text = json.dumps(content, ensure_ascii=False)
    
    for word in SENSITIVE_WORDS:
        if word in text:
            return 0  # 发现敏感词，直接0分
    
    return 20  # 无敏感词，满分
```

### 4.4 逻辑连贯性

```python
def check_coherence(content):
    score = 20
    
    # 检查排名是否合理（rating应该递减）
    ratings = [r.get('rating', 0) for r in content.get('recommendations', [])]
    if ratings != sorted(ratings, reverse=True):
        score -= 5
    
    # 检查价格与排名的关系（不应该只按价格排）
    prices = [parse_price(r.get('price', '')) for r in content.get('recommendations', [])]
    if prices == sorted(prices, reverse=True):
        score -= 5  # 如果完全按价格排，扣分
    
    # 检查是否有平替和免费方案
    has_alternative = any('平替' in r.get('one_liner', '') for r in content.get('recommendations', []))
    has_free = any('免费' in r.get('price', '') for r in content.get('recommendations', []))
    
    if not has_alternative:
        score -= 5
    if not has_free:
        score -= 5
    
    return max(score, 0)
```

### 5.1 翻译Prompt

```python
TRANSLATION_PROMPT = """
你是专业的AI工具评测翻译专家。
将以下中文推荐翻译成英文。

# 翻译原则

1. **保持专业术语**
   - "Prompt" 不翻译
   - "API" 不翻译
   - "Token" 不翻译

2. **符合英文习惯**
   - 不要直译
   - 使用地道表达

3. **保持结构**
   - JSON结构不变
   - 字段名不变

{chinese_content}

# 输出

翻译后的JSON（结构与原文相同）

def translate_to_english(chinese_content):
    prompt = TRANSLATION_PROMPT.format(
        chinese_content=json.dumps(chinese_content, ensure_ascii=False, indent=2)
    )
    
    response = claude_api.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=2000,
        messages=[
            {"role": "user", "content": prompt}
        ]
    )
    
    return parse_json(response.content[0].text)
```

```python
def check_translation_quality(chinese, english):
    # 检查结构是否一致
    if len(chinese.get('recommendations', [])) != len(english.get('recommendations', [])):
        return False
    
    # 检查关键字段是否存在
    for key in ['title', 'summary', 'recommendations']:
        if key not in english:
            return False
    
    # 检查是否有未翻译的中文
    text = json.dumps(english, ensure_ascii=False)
    chinese_chars = re.findall(r'[\u4e00-\u9fff]+', text)
    
    if len(chinese_chars) > 5:  # 允许少量中文（如品牌名）
        return False
    
    return True
```

### 6.2 批量生成

```python
def batch_generate(scenarios_data):
    """批量生成多个场景的推荐"""
    results = []
    
    # 第一次调用（建立缓存）
    first_result = generate_content(scenarios_data[0]['scenario'], scenarios_data[0]['data'])
    results.append(first_result)
    
    # 后续调用（使用缓存）
    for item in scenarios_data[1:]:
        result = generate_content(item['scenario'], item['data'])
        results.append(result)
        time.sleep(0.5)  # 避免限流
    
    return results
```

### 6.3 成本监控

```python
class CostMonitor:
    def __init__(self):
        self.total_input_tokens = 0
        self.total_output_tokens = 0
        self.cached_tokens = 0
    
    def track(self, response):
        usage = response.usage
        self.total_input_tokens += usage.input_tokens
        self.total_output_tokens += usage.output_tokens
        
        if hasattr(usage, 'cache_read_input_tokens'):
            self.cached_tokens += usage.cache_read_input_tokens
    
    def get_cost(self):
        input_cost = (self.total_input_tokens - self.cached_tokens) * 3 / 1_000_000
        output_cost = self.total_output_tokens * 15 / 1_000_000
        cache_savings = self.cached_tokens * 3 / 1_000_000
        
        return {
            "input_cost": input_cost,
            "output_cost": output_cost,
            "total_cost": input_cost + output_cost,
            "cache_savings": cache_savings
        }
```
